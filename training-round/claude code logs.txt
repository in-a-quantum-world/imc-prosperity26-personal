from datamodel import OrderDepth, TradingState, Order
from typing import Dict, List
import json


class Trader:
    POSITION_LIMITS = {
        "EMERALDS": 80,
        "TOMATOES": 80,
    }

    EMERALD_FAIR = 10000

    def run(self, state: TradingState):
        data = self._load_data(state.traderData)
        result: Dict[str, List[Order]] = {}

        for product, depth in state.order_depths.items():
            if product == "EMERALDS":
                result[product] = self.trade_emeralds(state, depth, data)
            elif product == "TOMATOES":
                result[product] = self.trade_tomatoes(state, depth, data)

        return result, 0, json.dumps(data, separators=(",", ":"))

    # -----------------------------
    # EMERALDS — unchanged (hard cap ~1050, not worth tuning)
    # -----------------------------
    def trade_emeralds(self, state: TradingState, depth: OrderDepth, data: dict) -> List[Order]:
        book = self._book(depth)
        if book is None:
            return []

        bb, bv1, ba, av1, mid = book
        pos = state.position.get("EMERALDS", 0)
        limit = self.POSITION_LIMITS["EMERALDS"]
        orders: List[Order] = []

        ema_key = "ema_emeralds"
        prev_ema = data.get(ema_key, self.EMERALD_FAIR)
        ema = 0.92 * prev_ema + 0.08 * mid
        data[ema_key] = ema

        imb1 = self._imbalance(bv1, av1)
        fair = self.EMERALD_FAIR + 1.2 * imb1 + 0.10 * (ema - self.EMERALD_FAIR)

        if ba <= self.EMERALD_FAIR and pos < limit:
            take = min(limit - pos, abs(depth.sell_orders.get(ba, 0)), 24)
            if take > 0:
                orders.append(Order("EMERALDS", ba, take))
                pos += take
        if bb >= self.EMERALD_FAIR and pos > -limit:
            take = min(limit + pos, depth.buy_orders.get(bb, 0), 24)
            if take > 0:
                orders.append(Order("EMERALDS", bb, -take))
                pos -= take

        orders += self._passive_mm(
            product="EMERALDS",
            best_bid=bb,
            best_ask=ba,
            fair=fair,
            pos=pos,
            limit=limit,
            timestamp=state.timestamp,
            base_size=68,
            back_size=16,
            signal=imb1,
            one_sided_threshold=0.30,
            max_bias=18,
        )
        return orders

    # -----------------------------
    # TOMATOES — enhanced with trend + OBI Z-score + selective taker
    # -----------------------------
    def trade_tomatoes(self, state: TradingState, depth: OrderDepth, data: dict) -> List[Order]:
        book = self._book(depth)
        if book is None:
            return []

        bb, bv1, ba, av1, mid = book
        pos = state.position.get("TOMATOES", 0)
        limit = self.POSITION_LIMITS["TOMATOES"]
        orders: List[Order] = []

        # --- Level-1 imbalance ---
        imb1 = self._imbalance(bv1, av1)

        # --- L2/L3 spoof signal (positive = fake buy pressure = bearish) ---
        bv2 = self._buy_vol(depth, 2)
        bv3 = self._buy_vol(depth, 3)
        av2 = self._sell_vol(depth, 2)
        av3 = self._sell_vol(depth, 3)
        l23imb = self._imbalance(bv2 + bv3, av2 + av3)
        spoof_signal = -l23imb  # positive -> bearish

        # --- EMA fair value ---
        ema_key = "ema_tomatoes"
        prev_ema = data.get(ema_key, mid)
        ema = 0.96 * prev_ema + 0.04 * mid
        data[ema_key] = ema

        # --- Trend slope: linear regression on recent 20 mids ---
        # positive slope = price trending up = bullish
        mids = data.get("tom_mids", [])
        mids.append(mid)
        if len(mids) > 20:
            mids = mids[-20:]
        data["tom_mids"] = mids

        trend_signal = 0.0
        n = len(mids)
        if n >= 8:
            x_mean = (n - 1) / 2.0
            y_mean = sum(mids) / n
            num = sum((i - x_mean) * (mids[i] - y_mean) for i in range(n))
            den = sum((i - x_mean) ** 2 for i in range(n))
            slope = num / den if den > 0 else 0.0
            # Normalise: 2 ticks/step is a strong trend → cap at ±1
            trend_signal = max(-1.0, min(1.0, slope / 2.0))

        # --- OBI Z-score: mean reversion of the imbalance itself ---
        # Extreme positive OBI (heavy buyers) → OBI will revert → slight bearish edge
        # Use Welford online update for rolling mean/variance
        obi_h = data.get("obi_hist", {"n": 0, "mean": 0.0, "m2": 0.0})
        n_obi = obi_h["n"] + 1
        delta = imb1 - obi_h["mean"]
        new_mean = obi_h["mean"] + delta / n_obi
        new_m2 = obi_h["m2"] + delta * (imb1 - new_mean)
        data["obi_hist"] = {"n": n_obi, "mean": new_mean, "m2": new_m2}

        if n_obi >= 30:
            std_obi = (new_m2 / n_obi) ** 0.5
            obi_z = (imb1 - new_mean) / max(std_obi, 0.01)
            # Clip Z to [-2, 2]; mean reversion → negative of Z
            obi_z_signal = -max(-2.0, min(2.0, obi_z)) / 2.0
        else:
            obi_z_signal = 0.0

        # --- Combined signal (bullish = positive) ---
        # Weights informed by Discord: trend+OBI reversal are the key unlocks above 2518
        combined = (
            0.55 * imb1
            + 1.10 * spoof_signal
            + 0.60 * trend_signal
            + 0.40 * obi_z_signal
        )

        # --- Fair value ---
        mean_revert = (ema - mid) / 8.0
        fair = mid + 1.2 * combined + 0.7 * mean_revert

        # --- Selective taker: only when trend + combined strongly agree ---
        # Spread is ~14 ticks; taking is only profitable when we predict a sustained
        # directional move. Require both trend_signal and combined to agree.
        TAKE_THRESHOLD = 0.40
        TAKE_SIZE = 12

        buy_cap = limit - pos
        sell_cap = limit + pos

        take_ok_buy = trend_signal > 0.12 and combined > TAKE_THRESHOLD
        take_ok_sell = trend_signal < -0.12 and combined < -TAKE_THRESHOLD

        if take_ok_buy and buy_cap > 0:
            take = min(TAKE_SIZE, buy_cap, abs(depth.sell_orders.get(ba, 0)))
            if take > 0:
                orders.append(Order("TOMATOES", ba, take))
                pos += take
                buy_cap -= take

        if take_ok_sell and sell_cap > 0:
            take = min(TAKE_SIZE, sell_cap, depth.buy_orders.get(bb, 0))
            if take > 0:
                orders.append(Order("TOMATOES", bb, -take))
                pos -= take
                sell_cap -= take

        # --- Passive MM ---
        orders += self._passive_mm(
            product="TOMATOES",
            best_bid=bb,
            best_ask=ba,
            fair=fair,
            pos=pos,
            limit=limit,
            timestamp=state.timestamp,
            base_size=68,
            back_size=26,
            signal=combined,
            one_sided_threshold=0.34,
            max_bias=22,
        )
        return orders

    # -----------------------------
    # Passive fill engine (unchanged from passive_fill)
    # -----------------------------
    def _passive_mm(
        self,
        product: str,
        best_bid: int,
        best_ask: int,
        fair: float,
        pos: int,
        limit: int,
        timestamp: int,
        base_size: int,
        back_size: int,
        signal: float,
        one_sided_threshold: float,
        max_bias: int,
    ) -> List[Order]:
        orders: List[Order] = []
        spread = best_ask - best_bid

        buy_cap = max(0, limit - pos)
        sell_cap = max(0, limit + pos)
        if buy_cap == 0 and sell_cap == 0:
            return orders

        if spread >= 3:
            join_bid = best_bid + 1
            join_ask = best_ask - 1
        else:
            join_bid = best_bid
            join_ask = best_ask

        inv_ratio = pos / max(1, limit)
        inv_shift = 0
        if inv_ratio > 0.75:
            inv_shift = 3
        elif inv_ratio > 0.50:
            inv_shift = 2
        elif inv_ratio > 0.25:
            inv_shift = 1
        elif inv_ratio < -0.75:
            inv_shift = -3
        elif inv_ratio < -0.50:
            inv_shift = -2
        elif inv_ratio < -0.25:
            inv_shift = -1

        fair_shift = 0
        if fair >= (best_bid + best_ask) / 2 + 1.5:
            fair_shift = 1
        elif fair <= (best_bid + best_ask) / 2 - 1.5:
            fair_shift = -1

        directional = 0
        if signal > 0.25:
            directional = 1
        elif signal < -0.25:
            directional = -1

        bid_px = join_bid + fair_shift - max(inv_shift, 0)
        ask_px = join_ask + fair_shift - min(inv_shift, 0)

        bid_px = min(bid_px, best_ask - 1)
        ask_px = max(ask_px, best_bid + 1)
        if bid_px >= ask_px:
            bid_px = min(join_bid, best_ask - 1)
            ask_px = max(join_ask, best_bid + 1)

        bias = int(max(-max_bias, min(max_bias, round(signal * 40))))
        bid_size = base_size + max(0, bias)
        ask_size = base_size + max(0, -bias)

        if pos > 0:
            bid_size -= int(abs(pos) * 0.60)
            ask_size += int(abs(pos) * 0.20)
        elif pos < 0:
            ask_size -= int(abs(pos) * 0.60)
            bid_size += int(abs(pos) * 0.20)

        if directional > 0:
            bid_size += 10
            ask_size -= 8
        elif directional < 0:
            ask_size += 10
            bid_size -= 8

        if timestamp >= 185000:
            if pos > 0:
                bid_size -= 20
                ask_size += 10
            elif pos < 0:
                ask_size -= 20
                bid_size += 10

        bid_size = max(0, min(buy_cap, bid_size))
        ask_size = max(0, min(sell_cap, ask_size))

        if abs(signal) >= one_sided_threshold:
            if signal > 0:
                ask_size = min(6, ask_size)
            else:
                bid_size = min(6, bid_size)

        if bid_size > 0:
            orders.append(Order(product, int(bid_px), int(bid_size)))
        if ask_size > 0:
            orders.append(Order(product, int(ask_px), -int(ask_size)))

        back_bid = max(best_bid, int(bid_px) - 1)
        back_ask = min(best_ask, int(ask_px) + 1)

        back_bid_size = max(0, min(buy_cap - bid_size, back_size))
        back_ask_size = max(0, min(sell_cap - ask_size, back_size))

        if timestamp >= 190000:
            back_bid_size = 0 if pos > 0 else back_bid_size
            back_ask_size = 0 if pos < 0 else back_ask_size

        if back_bid_size > 0 and back_bid < best_ask:
            orders.append(Order(product, int(back_bid), int(back_bid_size)))
        if back_ask_size > 0 and back_ask > best_bid:
            orders.append(Order(product, int(back_ask), -int(back_ask_size)))

        return orders

    # -----------------------------
    # Helpers
    # -----------------------------
    def _load_data(self, trader_data: str) -> dict:
        if not trader_data:
            return {}
        try:
            return json.loads(trader_data)
        except Exception:
            return {}

    def _book(self, depth: OrderDepth):
        if not depth.buy_orders or not depth.sell_orders:
            return None
        best_bid = max(depth.buy_orders.keys())
        best_ask = min(depth.sell_orders.keys())
        bid_vol = depth.buy_orders[best_bid]
        ask_vol = abs(depth.sell_orders[best_ask])
        mid = (best_bid + best_ask) / 2
        return best_bid, bid_vol, best_ask, ask_vol, mid

    def _imbalance(self, buy_vol: float, sell_vol: float) -> float:
        denom = buy_vol + sell_vol
        if denom <= 0:
            return 0.0
        return (buy_vol - sell_vol) / denom

    def _buy_vol(self, depth: OrderDepth, level: int) -> int:
        prices = sorted(depth.buy_orders.keys(), reverse=True)
        if len(prices) < level:
            return 0
        return int(depth.buy_orders[prices[level - 1]])

    def _sell_vol(self, depth: OrderDepth, level: int) -> int:
        prices = sorted(depth.sell_orders.keys())
        if len(prices) < level:
            return 0
        return int(abs(depth.sell_orders[prices[level - 1]]))

from prosperity3bt.datamodel import OrderDepth, TradingState, Order


class Trader:
    def run(self, state: TradingState):
        result = {}

        for product in state.order_depths:
            order_depth: OrderDepth = state.order_depths[product]
            orders = []

            if len(order_depth.sell_orders) > 0:
                best_ask = min(order_depth.sell_orders.keys())
                best_ask_volume = order_depth.sell_orders[best_ask]
                # Buy at the best ask if it seems cheap
                if product == "EMERALDS" and best_ask < 10000:
                    orders.append(Order(product, best_ask, -best_ask_volume))
                elif product == "TOMATOES" and best_ask < 5006:
                    orders.append(Order(product, best_ask, -best_ask_volume))

            if len(order_depth.buy_orders) > 0:
                best_bid = max(order_depth.buy_orders.keys())
                best_bid_volume = order_depth.buy_orders[best_bid]
                # Sell at the best bid if it seems expensive
                if product == "EMERALDS" and best_bid > 10000:
                    orders.append(Order(product, best_bid, -best_bid_volume))
                elif product == "TOMATOES" and best_bid > 5006:
                    orders.append(Order(product, best_bid, -best_bid_volume))

            result[product] = orders

Yes, exactly. There are two separate tools here — let me clarify what each one does and what the full pipeline looks like.

The Two Tools
1. discord_reader.py — offline reader for exported JSON files
Reads static JSON exports you've already downloaded. CLI for browsing/searching them.

Pipeline:


DiscordChatExporter (external GUI/CLI app)
  → exports JSON files to tools/exports/
  → discord_reader.py reads those files
You already have exports in tools/exports/ covering up to April 11. To get newer messages you'd re-export manually.

Usage:


# Browse everything
python tools/discord_reader.py --dir tools/exports/

# Only the algo-trading channel, last 50 msgs
python tools/discord_reader.py --dir tools/exports/ --channel algo-trading --limit 50

# Search for a term across all channels
python tools/discord_reader.py --dir tools/exports/ --search "moon rock"

# Only messages since a date
python tools/discord_reader.py --dir tools/exports/ --since 2026-04-10

# Filter by user
python tools/discord_reader.py --dir tools/exports/ --user MarkBrezina
2. discord_intel.py — live Discord bot with Claude analysis
This is the real-time pipeline. Requires a bot token and runs continuously.

Pipeline:


Bot joins Discord server (needs token)
  → saves messages to SQLite as they arrive
  → every 120s: sends batches of 30 msgs to Claude API
  → Claude outputs: VERIFIED INTEL / MISINFORMATION / COMPETITOR FOCUS / ACTIONS
  → stored reports readable with --report
Setup (one time):


tools\set_env.bat   ← paste your DISCORD_BOT_TOKEN and ANTHROPIC_API_KEY
python tools\discord_intel.py   ← starts live monitoring
Backfill historical data from existing exports — this isn't directly wired up yet. The bot's backfill mode (--backfill N) pulls from its own SQLite, not your JSON exports.

Recommended Pipeline Going Forward
Since re-exporting manually is tedious, the most practical setup is:

Live bot for new messages — run discord_intel.py from now on; it captures everything going forward automatically
Existing JSON exports — use discord_reader.py to search what you already have (April 11 and earlier)
The gap (today is April 12) means you missed ~1 day. You could re-export once via DiscordChatExporter to get that, or just start the live bot now and accept the small gap.

The analysis I ran earlier (the intel report) used discord_reader.py directly on the JSON files — that's the same thing you'd do for historical queries. No "uploading to a Python file" needed, just point --dir at your exports folder.

# Discord Intelligence (through 2026-04-11)

## MarkBrezina's confirmed framework (most active top competitor)
1. Market make at each T
2. Look for take opportunities and take at T
3. Consider inventory position, target inv and risk at T, adjust
4. Execute orders with mm, mt, tf logic

## Confirmed strategy facts
- **2,518 "soft cap"**: Achievable with pure passive MM (bid+1, ask-1, both products, max size). Any strategy that beats this requires trend/drift detection + taking.
- **3,000+ barrier**: Achievable without overfitting via inventory skew + trend following + OBI signal. MarkBrezina confirmed 2.9k with no overfit.
- **Emeralds max**: ~1,050. Multiple competitors confirmed. Not worth further optimization.
- **Tomatoes dynamics**: Non-pure-random-walk (ACF/PACF/Jarque-Bera confirmed). Has trend drift. Adding linreg slope + EMA trend detection is the key unlock.
- **Fill volume bottleneck**: Simulator caps fills by bot activity. Increasing our order size beyond available bot volume does nothing extra. Never reduce size with inventory penalty on EMERALDS.
- **Tomas (admin) on Sharpe**: "I'd say the main difference between 2.7k and 3k is about 300" — and consistency/Sharpe matters for the 1M timestamp real run.

## Trend detection (from MarkBrezina)
- "Trend detection can be linear regression, combined with EWMA or SMA"
- "amogos: applied the trend drift thing and it worked, went above the 2518 soft cap"
- Positive trend → sell more (take), negative trend → buy more (take)
- Signal strength metric important: "how likely is this trend to continue"

## OBI / order book signals
- Level 1 imbalance: direction signal (buy pressure → price up)
- Levels 2-3 imbalance: "spoof signal" — heavy L2/L3 buy side is usually fake (bearish)
- OBI Z-score: extreme OBI mean-reverts → secondary contrarian signal
- incognitoman ran ACF/PACF on Tomatoes: confirms non-random-walk properties

# Coding Style Preferences

## Keep proven signal weights unchanged when adding new signals
Don't reduce existing calibrated weights (e.g., `0.65*imb1 + 1.35*spoof`) when introducing new features. Instead, add new terms additively with small weights. Reducing proven weights breaks calibration.

**Why:** Reducing imb1 from 0.65→0.55 and spoof from 1.35→1.10 broke the trader — had to revert.
**How to apply:** Any new signal should be added as `+ 0.X * new_signal` without touching existing weights.

## Never add a taker block to Tomatoes without confirming it helps
The taker block was firing at a 14-tick spread and losing money. Removed it entirely.

**Why:** Taker logic needs precise threshold calibration or it destroys the passive MM edge.
**How to apply:** Only add taker logic if it's been confirmed profitable in backtests. Start with conservative thresholds.

## Run both Kevin's and MC backtester before declaring a change an improvement
Kevin's is deterministic, MC captures variance. A change must improve at least one without hurting both.

**Why:** Conflicting signals between the two helped catch overfit strategies.
**How to apply:** Always report both scores when evaluating changes.

## Use Python 3.12 for backtesting
The working environment has Python 3.12 and 3.14 installed. Use 3.12 for the backtester (the one with `python` in PATH on this machine).

**Why:** Interpreter mismatch caused confusion about which file scored what.
**How to apply:** Always use `python` (not `python3`) for backtester commands.
